        "\n",
        "import gradio as gr\n",
        "import cv2\n",
        "from ultralytics import YOLO\n",
        "from collections import Counter\n",
        "\n",
        "# Load YOLO model\n",
        "model = YOLO(\"yolov8n.pt\")\n",
        "\n",
        "# Object detection function\n",
        "def detect_objects(image):\n",
        "    if image is None:\n",
        "        return None, \"Please upload an image.\"\n",
        "\n",
        "    # Convert RGB image to BGR\n",
        "    image_bgr = cv2.cvtColor(image, cv2.COLOR_RGB2BGR)\n",
        "\n",
        "    # Detect objects\n",
        "    results = model(image_bgr)\n",
        "\n",
        "    # Draw bounding boxes\n",
        "    output = results[0].plot()\n",
        "\n",
        "    # Convert output to RGB\n",
        "    output = cv2.cvtColor(output, cv2.COLOR_BGR2RGB)\n",
        "\n",
        "    # Get detected object names\n",
        "    names = results[0].names\n",
        "    detected = []\n",
        "\n",
        "    for box in results[0].boxes:\n",
        "        class_id = int(box.cls[0])\n",
        "        confidence = float(box.conf[0])\n",
        "        object_name = names[class_id]\n",
        "\n",
        "        detected.append(\n",
        "            f\"{object_name} - {confidence * 100:.2f}%\"\n",
        "        )\n",
        "\n",
        "    if detected:\n",
        "        summary = \"Detected Objects:\\n\\n\"\n",
        "        for item in detected:\n",
        "            summary += \"• \" + item + \"\\n\"\n",
        "    else:\n",
        "        summary = \"No objects detected.\"\n",
        "\n",
        "    return output, summary\n",
        "\n",
        "\n",
        "# Create Gradio interface\n",
        "with gr.Blocks(title=\"AI Object Detection\") as demo:\n",
        "\n",
        "    gr.Markdown(\"# AI-Based Object Detection\")\n",
        "    gr.Markdown(\n",
        "        \"Upload an image to detect objects using YOLO.\"\n",
        "    )\n",
        "\n",
        "    input_image = gr.Image(\n",
        "        label=\"Upload Image\",\n",
        "        type=\"numpy\"\n",
        "    )\n",
        "\n",
        "    detect_button = gr.Button(\n",
        "        \"Detect Objects\",\n",
        "        variant=\"primary\"\n",
        "    )\n",
        "\n",
        "    output_image = gr.Image(\n",
        "        label=\"Detection Result\"\n",
        "    )\n",
        "\n",
        "    output_text = gr.Textbox(\n",
        "        label=\"Detected Object Details\",\n",
        "        lines=8\n",
        "    )\n",
        "\n",
        "    detect_button.click(\n",
        "        fn=detect_objects,\n",
        "        inputs=input_image,\n",
        "        outputs=[output_image, output_text]\n",
        "    )\n",
        "\n",
        "demo.launch(share=True)"
